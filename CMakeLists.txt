cmake_minimum_required(VERSION 3.16)
project(qiyu_head_tracking LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(QIYU_DEVICE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../qiyu-device-service" CACHE PATH "Device source checkout")
add_subdirectory("${QIYU_DEVICE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/device")
file(GLOB_RECURSE HEXAGON_SOURCES CONFIGURE_DEPENDS "third_party/hexagonrpc/libhexagonrpc/*.c")
add_library(qiyu_hexagonrpc STATIC ${HEXAGON_SOURCES})
set_target_properties(qiyu_hexagonrpc PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(qiyu_hexagonrpc PUBLIC third_party/hexagonrpc/include)
foreach(kind executable plugin)
  if(kind STREQUAL executable)
    add_executable(a081-qvr-stock-live-stream src/main.cpp src/qvr_bridge.cpp)
    set(target a081-qvr-stock-live-stream)
  else()
    add_library(a081-qvr-client-live-stream SHARED src/main.cpp src/qvr_bridge.cpp)
    set(target a081-qvr-client-live-stream)
    target_compile_definitions(${target} PRIVATE A081_QVR_PLUGIN)
  endif()
  target_include_directories(${target} PRIVATE include src)
  target_compile_options(${target} PRIVATE -O2 -Wall -Wextra)
  target_link_libraries(${target} PRIVATE qiyu_device_abi qiyu_hexagonrpc pthread dl)
endforeach()
add_executable(a081-tracking-observe src/tracking-observe.c)
target_include_directories(a081-tracking-observe PRIVATE include)

target_compile_definitions(a081-tracking-observe PRIVATE _GNU_SOURCE)
